  import { chordDictionaryService } from './src/services/chordDictionaryService';
  
  // Obtener todas las posiciones de un acorde

  const position = chordDictionaryService.getChordPosition('Cmaj7', 0);

  const hasMultiple = chordDictionaryService.hasMultiplePositions('Am');

🔄 CÓMO FUNCIONA INTERNAMENTE
═══════════════════════════════════════════════════════════════════════════════

1. DEFINICIÓN DE FÓRMULAS
   • Se define cada tipo de acorde como intervalo de semitonos
   • Ejemplo: Mayor = [0, 4, 7] (1-3-5)

2. GENERACIÓN DE POSICIONES
   Para cada acorde:
   • Se buscan las notas del acorde en diferentes trastes
   • Se buscan hasta 5 posiciones óptimas
   • Se valida que contiene todas las notas requeridas

3. CLASIFICACIÓN DE DIFICULTAD
   • Basada en número de trastes, cejillas y posición en el diapasón
   • Easy: posiciones abiertas, sin cejilla
   • Medium: cejilla o trastes medios
   • Hard: cejilla + trastes altos

4. GENERACIÓN DE METADATOS
   • Versión del diccionario
   • Fecha de generación
   • Estadísticas (total acordes, posiciones, etc.)
   • Información de categorías
